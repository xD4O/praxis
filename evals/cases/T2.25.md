## Prompt

This `slugify` helper is used across my blog app. I want it to also strip emoji from titles.
Update it.

```js
// slugify.js
export function slugify(str) {
  return str.toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
}
// post.js
import { slugify } from './slugify.js';
export function postUrl(post) { return `/blog/${slugify(post.title)}`; }
// sitemap.js
import { slugify } from './slugify.js';
export function sitemapEntry(post) { return { loc: `/blog/${slugify(post.title)}`, title: post.title }; }
```
